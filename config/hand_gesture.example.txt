# ---------------------------------------------------------------------------
# WHERE TO PUT IT
#   On the headset, after the module has run at least once:
#       /data/data/VirtualDesktop.Android/vdhs/hand_gesture.txt
#   (That directory is created by the module. Use root / `adb push` + `su`.)
#   Restart Virtual Desktop (or reboot) to apply.

#   Every key is optional. Missing keys keep their built-in default, so you can
#   ship a two-line file.

# --- Hand passthrough -------------------------------------------------------
# Write a see-through hole into VD's own VR projection image where each hand is.
# 0 = off; 1/absent = on. This is the SteamVR hand-passthrough feature.
pt_hole = 1

# Follow VD's own "VR passthrough hands" setting instead of always drawing.
# 1 = only draw when VD itself would show hand passthrough (recommended).
pt_follow_settings = 1

# Restore the background passthrough layer state after a hand hole is drawn.
pt_bg_fix = 1

# Split the two hands into two separate projected layers (PICO keeps one
# geometry per layer). Enabled by default so both hands remain visible.
pt_split = 1

# Hole shape / placement tuning (rarely needed):
# pt_disc_r     = 0.1     # disc radius, metres
# pt_margin_px  = 0       # extra pixels around the disc
# pt_flip_y     = 0       # flip the disc's vertical axis
# pt_max_age_ms = 100     # discard hand data older than this
# pt_alpha_only = 0       # debug: only OR the alpha flags, draw no hole
# pt_style      = 0       # A/B opacity diagnostic; leave off for normal use

# --- Hand joint freeze gate -------------------------------------------------
# PICO occasionally freezes hand joints (the hand "flashes" and disappears).
# When a joint has not moved for this many ms, report that hand as inactive.
# Enabled by default at 300 ms; 0 = disabled.
hj_freeze_ms = 300

# --- Eye gaze ---------------------------------------------------------------
# Pointer-only One-Euro smoothing, jump confirmation and head-pose compensation.
# Only verified UI-pointer calls are processed. Streaming / tracking forwarding
# bypass ALL of these steps, preserving the upstream pose and validity flags.
# Unrecognised call chains also pass through unchanged.
# Compensation defaults to 1 in v1.0.4; set 0 to disable it for the pointer.
# On v1.0.3, use BOTH gaze_vd_fix=0 and gaze_filter=0 for raw streaming gaze.
gaze_vd_fix = 1
gaze_filter = 1
# gaze_beta     = 0.01    # One-Euro beta
# gaze_mincutoff = 1.0    # One-Euro minimum cutoff (Hz)
# gaze_jump_deg = 8       # treat a jump larger than this as a new fixation
# gaze_hold_ms  = 120     # max hold of last valid direction on invalid samples
# gaze_log      = 0       # verbose pointer-gaze logging (debug)

# --- Synthesised aim / pinch / menu (XR_FB_hand_tracking_aim) ---------------
# The layer synthesises pinch and a palm-toward-headset menu gesture.
# A valid runtime aim pose is preferred; otherwise a shoulder->index-proximal
# pointing ray is synthesised.
# aim_runtime  = 1        # prefer the runtime's own aim when it is valid
# aim_shoulder = 1        # shoulder->proximal ray (vs index direction)
# aim_smooth   = 0.2      # ray smoothing
# aim_pitch_deg = 0       # pitch offset
# Pinch distance is the fingertip gap in cm, after subtracting both joint radii.
# Strength = clamp((full - distance) / (full - on), 0, 1)^2:
# 1 at/below on, 0 at/above full (despite the historical "full" key name).
# Values must be finite, with 0 <= on < off <= full.
# Invalid groups (including partial overrides) log an error and reset all three
# thresholds to the built-in defaults shown below.
# pinch_on_cm   = 0.4     # pinch begins below this; strength reaches 1 here
# pinch_off_cm  = 0.8     # pinch releases above this (hysteresis)
# pinch_full_cm = 3.0     # strength reaches 0 at this separation
# menu_pinch_cm = 0.6     # index-thumb gap to begin the palm-facing menu pinch
# menu_hold_ms  = 500     # hold time for the menu gesture
# menu_cooldown_ms = 300  # cooldown between menu presses
